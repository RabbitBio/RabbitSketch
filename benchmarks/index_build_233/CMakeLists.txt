cmake_minimum_required(VERSION 3.16)
project(index_build_233 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CXXAPI ON CACHE BOOL "" FORCE)
set(RABBITSKETCH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RABBITSKETCH_NATIVE_ARCH ON CACHE BOOL "" FORCE)
add_compile_options(-g1 -funroll-loops -fno-lto)
add_subdirectory(../.. rabbitsketch EXCLUDE_FROM_ALL)
find_package(OpenMP REQUIRED)
find_library(DEFLATE_LIBRARY deflate REQUIRED)
add_executable(index_build_233 runner.cpp)
target_compile_options(index_build_233 PRIVATE -march=native)
target_link_libraries(index_build_233 PRIVATE rabbitsketch_static OpenMP::OpenMP_CXX ${DEFLATE_LIBRARY})
add_executable(output_digest digest.cpp)
target_include_directories(output_digest PRIVATE ../../src)
target_link_libraries(output_digest PRIVATE OpenMP::OpenMP_CXX)
