cmake_minimum_required(VERSION 3.16)
project(rabbitsketch_screening LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(RABBITSKETCH_SOURCE "" CACHE PATH "Pinned RabbitSketch 2 source checkout")
if(NOT EXISTS "${RABBITSKETCH_SOURCE}/src/fracminhash.h")
  message(FATAL_ERROR "Set RABBITSKETCH_SOURCE to the RabbitSketch 2 checkout; see README.md")
endif()
set(CXXAPI ON CACHE BOOL "" FORCE)
set(RABBITSKETCH_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${RABBITSKETCH_SOURCE}" rabbitsketch EXCLUDE_FROM_ALL)
add_executable(synthetic_pilot synthetic_pilot.cpp)
target_link_libraries(synthetic_pilot PRIVATE rabbitsketch_static)
if(EXISTS "${RABBITSKETCH_SOURCE}/src/estimators/PartitionedSetSketch.h")
  enable_testing()
  add_executable(test_partitioned_setsketch test_partitioned_setsketch.cpp)
  target_link_libraries(test_partitioned_setsketch PRIVATE rabbitsketch_static)
  add_test(NAME partitioned_setsketch_model COMMAND test_partitioned_setsketch)
endif()
option(RABBITSKETCH_SCREENING_MAIN "Build the existing CLI with the synthetic batch adapter" OFF)
if(RABBITSKETCH_SCREENING_MAIN)
  find_package(OpenMP REQUIRED)
  find_library(SCREENING_DEFLATE_LIBRARY deflate REQUIRED)
  add_executable(rabbitsketch_pilot
    "${RABBITSKETCH_SOURCE}/examples/main.cpp"
    "${RABBITSKETCH_SOURCE}/examples/sub_command.cpp"
    synthetic_pilot.cpp main_batch.cpp ../regimes/main_regimes.cpp)
  target_include_directories(rabbitsketch_pilot PRIVATE "${RABBITSKETCH_SOURCE}/examples")
  target_compile_definitions(rabbitsketch_pilot PRIVATE RABBITSKETCH_SCREENING_EMBEDDED)
  target_link_libraries(rabbitsketch_pilot PRIVATE rabbitsketch_static OpenMP::OpenMP_CXX ${SCREENING_DEFLATE_LIBRARY})
endif()
