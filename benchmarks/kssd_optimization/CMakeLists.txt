cmake_minimum_required(VERSION 3.16)
project(kssd_optimization LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
set(CXXAPI ON CACHE BOOL "" FORCE)
set(RABBITSKETCH_NATIVE_ARCH ON CACHE BOOL "" FORCE)
set(RABBITSKETCH_BUILD_TESTS ON CACHE BOOL "")
set(RABBITSKETCH_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Library source snapshot")
add_compile_options(-g1 -funroll-loops -fno-lto)
add_subdirectory(${RABBITSKETCH_SOURCE} rabbitsketch)
find_package(OpenMP REQUIRED)
find_library(DEFLATE_LIBRARY deflate REQUIRED)
add_executable(rabbitsketch_main ${RABBITSKETCH_SOURCE}/examples/main.cpp ${RABBITSKETCH_SOURCE}/examples/sub_command.cpp)
target_compile_options(rabbitsketch_main PRIVATE -march=native)
target_link_libraries(rabbitsketch_main PRIVATE rabbitsketch_static OpenMP::OpenMP_CXX ${DEFLATE_LIBRARY})
enable_testing()
add_executable(kssd_bench bench.cpp)
target_compile_options(kssd_bench PRIVATE -march=native)
target_link_libraries(kssd_bench PRIVATE rabbitsketch_static OpenMP::OpenMP_CXX)
